<script id="fs" type="x-shader/x-fragment">#version 300 es
precision highp float;
out vec4 outColor;
uniform vec2 R;
uniform float T;
uniform vec2 M;

float hash(vec2 p){p=fract(p*vec2(123.34,456.21));p+=dot(p,p+45.32);return fract(p.x*p.y);}
float noise(vec2 p){
  vec2 i=floor(p),f=fract(p);f=f*f*(3.-2.*f);
  return mix(mix(hash(i),hash(i+vec2(1,0)),f.x),mix(hash(i+vec2(0,1)),hash(i+vec2(1,1)),f.x),f.y);
}
float sdBox(vec3 p,vec3 b,float r){
  vec3 q=abs(p)-b+r;
  return length(max(q,0.))+min(max(q.x,max(q.y,q.z)),0.)-r;
}

const vec3 L=vec3(-.5922,.5004,.6310);          // key light direction (normalised below as well)
const vec3 LC=vec3(1.,.965,.91);

vec2 map(vec3 p){
  vec2 res=vec2(p.y,1.);
  // A — monumental slab, left
  float dA=sdBox(p-vec3(-3.6,4.4,-0.8),vec3(1.35,4.4,.62),.05);
  if(dA<res.x)res=vec2(dA,2.);
  // B — mid slab, set back, offset
  float dB=sdBox(p-vec3(.3,3.1,-3.0),vec3(1.05,3.1,.55),.05);
  if(dB<res.x)res=vec2(dB,3.);
  // C — long low graphite plinth, foreground right
  float dC=sdBox(p-vec3(3.5,.75,.9),vec3(2.0,.75,1.05),.05);
  if(dC<res.x)res=vec2(dC,4.);
  return res;
}
vec3 calcN(vec3 p){
  vec2 e=vec2(1.,-1.)*.0007;
  return normalize(e.xyy*map(p+e.xyy).x+e.yyx*map(p+e.yyx).x+e.yxy*map(p+e.yxy).x+e.xxx*map(p+e.xxx).x);
}
float softShadow(vec3 ro,vec3 rd){
  float res=1.,t=.06;
  for(int i=0;i<44;i++){
    float h=map(ro+rd*t).x;
    res=min(res,10.*h/t);
    t+=clamp(h,.03,.7);
    if(res<.001||t>24.)break;
  }
  res=clamp(res,0.,1.);
  return res*res*(3.-2.*res);
}
float calcAO(vec3 p,vec3 n){
  float o=0.,s=1.;
  for(int i=0;i<5;i++){
    float h=.03+.14*float(i);
    o+=(h-map(p+n*h).x)*s;s*=.82;
  }
  return clamp(1.-1.7*o,0.,1.);
}
vec2 trace(vec3 ro,vec3 rd){
  float t=.02;
  for(int i=0;i<110;i++){
    vec2 h=map(ro+rd*t);
    if(h.x<.0006*t)return vec2(t,h.y);
    t+=h.x;
    if(t>46.)break;
  }
  return vec2(-1.,0.);
}

vec3 env(vec3 rd){
  vec3 Ln=normalize(L);
  vec3 c=mix(vec3(.004,.005,.007),vec3(.022,.027,.034),smoothstep(-.25,.95,rd.y));
  c+=vec3(1.,.97,.92)*smoothstep(.80,.93,dot(rd,Ln))*4.;           // big softbox
  c+=vec3(.72,.82,1.)*smoothstep(.962,.985,dot(rd,normalize(vec3(.82,.28,-.2))))*2.2; // cool strip light
  c+=vec3(.11,.13,.17)*exp(-length(rd.xy-vec2(.10,.26))*3.4)*step(rd.z,0.); // backlit glow
  return c;
}

vec3 shadeObj(vec3 p,vec3 n,vec3 rd,float id,bool full){
  vec3 Ln=normalize(L);
  vec3 alb=id<2.5?vec3(.15,.16,.18):(id<3.5?vec3(.22,.23,.25):vec3(.06,.065,.072));
  float sh=full?softShadow(p+n*.012,Ln):1.;
  float spot=smoothstep(15.,3.5,length(p-vec3(-6.,6.,5.5)));
  float ao=full?calcAO(p,n):.75;
  float dif=clamp(dot(n,Ln),0.,1.)*sh*(.12+.88*spot);
  float streak=noise(vec2(p.y*210.,(p.x+p.z)*1.7));
  streak=.55*streak+.45*noise(vec2(p.y*37.,(p.x+p.z)*.8));
  vec3 col=alb*LC*dif*2.6;
  col+=alb*mix(vec3(.006,.007,.009),vec3(.03,.036,.045),n.y*.5+.5)*ao*1.4;
  vec3 h=normalize(Ln-rd);
  float sp=pow(max(dot(n,h),0.),24.+90.*streak)*dif;
  col+=LC*sp*(.7+.9*streak);
  float fr=.05+.95*pow(1.-max(dot(n,-rd),0.),5.);
  col+=env(reflect(rd,n))*fr*(.30+.9*streak)*ao*(id>3.5?.6:1.)*(.25+.75*spot);
  // crisp edge highlight where the chamfer catches the key light
  float rim=pow(1.-max(dot(n,-rd),0.),3.)*.22;
  col+=LC*rim*dif;
  return col;
}

void main(){
  vec2 uv=(gl_FragCoord.xy*2.-R)/R.y;
  vec3 ro=vec3(sin(T*.07)*.38+M.x*.55,.85+M.y*.2,8.4);
  vec3 ta=vec3(-.2,2.5,0.);
  vec3 cw=normalize(ta-ro),cu=normalize(cross(cw,vec3(0,1,0))),cv=cross(cu,cw);
  vec3 rd=normalize(uv.x*cu+uv.y*cv+1.85*cw);
  vec3 Ln=normalize(L);

  vec2 hit=trace(ro,rd);
  float t=hit.x;
  vec3 col;
  if(t<0.){
    col=env(rd);
    t=46.;
  }else{
    vec3 p=ro+rd*t;
    if(hit.y<1.5){
      // polished concrete floor
      vec3 n=vec3(0,1,0);
      n.xz+=(vec2(noise(p.xz*1.1),noise(p.xz*1.1+17.))-.5)*.008;
      n=normalize(n);
      float ndv=max(dot(n,-rd),0.);
      float fr=.10+.90*pow(1.-ndv,4.);
      float sh=softShadow(p+n*.01,Ln);
      vec3 base=vec3(.016,.017,.020);
      col=base*LC*max(dot(n,Ln),0.)*sh*2.2+base*.5;
      vec3 r=reflect(rd,n);
      vec2 h2=trace(p+n*.02,r);
      vec3 rc;
      if(h2.x<0.)rc=env(r);
      else{
        vec3 q=p+n*.02+r*h2.x;
        rc=shadeObj(q,calcN(q),r,h2.y,false);
        rc=mix(rc,vec3(.04,.048,.06),1.-exp(-h2.x*.05));
      }
      col+=rc*fr*.7*exp(-h2.x*.05);
      // soft pool of light on the floor
      col+=LC*exp(-length((p.xz-vec2(-3.,3.5))*vec2(.20,.28)))*.05*sh;
    }else{
      col=shadeObj(p,calcN(p),rd,hit.y,true);
    }
  }

  // atmosphere
  float dist=min(t,46.);
  vec3 haze=vec3(.006,.008,.011)+vec3(.55,.60,.70)*pow(max(dot(rd,Ln),0.),6.)*.01;
  col=mix(col,haze,1.-exp(-dist*.028));

  // light shafts (cheap single scattering through the gaps)
  float j=hash(gl_FragCoord.xy+fract(T)*13.)*.5;
  float acc=0.;
  float maxS=min(dist,22.);
  for(int i=0;i<18;i++){
    float s=(float(i)+j)*maxS/18.;
    vec3 q=ro+rd*s;
    float v=1.;
    for(int k=1;k<7;k++){
      if(map(q+Ln*float(k)*.95).x<.04){v=0.;break;}
    }
    acc+=v;
  }
  acc/=18.;
  col+=LC*acc*(.022+.11*pow(max(dot(rd,Ln),0.),3.))*smoothstep(0.,6.,maxS);

  // filmic grade
  col*=.9;
  col=(col*(2.51*col+.03))/(col*(2.43*col+.59)+.14);
  col=pow(clamp(col,0.,1.),vec3(1./2.2));
  // vignette + grain
  vec2 s=gl_FragCoord.xy/R;
  col*=mix(.35,1.,pow(16.*s.x*s.y*(1.-s.x)*(1.-s.y),.30));
  col+=(hash(gl_FragCoord.xy+T)-.5)*.018;
  outColor=vec4(col,1.);
}
